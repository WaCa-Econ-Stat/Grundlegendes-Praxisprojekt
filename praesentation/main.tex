\section{Überblick und Terminologie}

\begin{frame}{Was bedeutet extreme Armut?}
  \begin{itemize}[<+->]
    \setlength{\itemsep}{4mm}
    \item Die Armutsquote misst den Anteil der Menschen, die mit weniger als \textbf{3 internationalen Dollar pro Person und Tag} leben.
    \item Internationale Dollar berücksichtigen Unterschiede in den Lebenshaltungskosten. Die Grenze bezieht sich auf die Kaufkraft von 2021.
    \item Je nach Land und Erhebung basiert die Quote auf dem \textbf{Einkommen} oder dem \textbf{Konsum} der Haushalte.
  \end{itemize}
  \note{\QuelleArmut}
\end{frame}

\begin{frame}{Datengrundlage}
  \begin{itemize}[<+->]
    \setlength{\itemsep}{4mm}
    \item \textbf{Welt und Regionen.} Modellierte Armutsquoten für 1990 bis 2024.
    \item \textbf{Länder.} Beobachtungen aus nationalen Erhebungen. Für die Fallauswahl vergleichen wir Reihen mit derselben Erhebungsart und Vergleichbarkeitskennung.
    \item \textbf{Kindersterblichkeit.} Todesfälle vor dem fünften Geburtstag je 100 Lebendgeburten. Wir verbinden Armut und Kindersterblichkeit aus demselben Länderjahr.
  \end{itemize}
  \note{\QuelleArmut\space\QuelleGesundheit}
\end{frame}

\section{Zusammenfassung und Diskussion}
\Agenda[currentsection]

\begin{frame}{Zusammenfassung}
  \begin{itemize}[<+->]
    \setlength{\itemsep}{4mm}
    \item Die weltweite Armutsquote ging zurück. In den Regionen fiel der Rückgang unterschiedlich stark aus.
    \item In 18 von 19 Ländern mit vergleichbaren Jahreswerten war die Armutsquote mit Einkommensdaten im Median höher.
    \item Länder mit mehr extremer Armut wiesen häufig eine höhere Kindersterblichkeit auf. Simbabwe zeigt, dass sich beide Größen über die Zeit unterschiedlich entwickeln können.
  \end{itemize}
\end{frame}

\begin{frame}{Diskussion}
  \begin{itemize}
    \setlength{\itemsep}{5mm}
    \item Welche Armutsgrenze ist für den Vergleich welcher Länder aussagekräftig?
    \item Wie verändern Erhebungsart und fehlende Daten das Bild einer Region?
    \item Welche Informationen zu Gesundheit und Versorgung würden helfen, den Zusammenhang mit Armut genauer einzuordnen?
  \end{itemize}
\end{frame}
